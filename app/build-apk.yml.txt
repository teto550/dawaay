name: Build Android APK

on:
  push:
    paths:
      - 'app/**'
      - '.github/workflows/build-apk.yml'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install and sync
        working-directory: app
        run: |
          npm ci
          npx cap sync android

      - name: Build debug APK
        working-directory: app/android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon --stacktrace

      - name: Collect APK
        run: cp app/android/app/build/outputs/apk/debug/app-debug.apk dawaay.apk

      - uses: actions/upload-artifact@v4
        with:
          name: dawaay-apk
          path: dawaay.apk

      - name: Publish release
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          gh release delete apk-latest --yes --cleanup-tag || true
          gh release create apk-latest dawaay.apk --title "دوائي - تطبيق أندرويد" --notes "حمّل dawaay.apk وثبّته (اسمح بالتثبيت من مصادر غير معروفة)."
